Pitcher API Demo

List files, open them, or pick from a native modal.

The Code

// 1. Initialize
const api = pitcher.useApi()
const env = await api.getEnv()

// 2. List files
const { results, count } = await api.getFiles({
  instance_id: env.pitcher.instance.id,
  page_size: 10
})

// 3. Open a file
api.open({ fileId: file.id })

// 4. Select a file via native modal
const { user_action, content } = await api.selectContent()
// user_action: 'selected' | 'cancelled'
// content: array of { id, name, type, ... }
